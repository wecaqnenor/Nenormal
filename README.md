        super().__init__(parent=parent)
        self.view = QWidget(self)
        self.layout = QVBoxLayout(self.view)

        self.search_bar = SearchLineEdit(self)
        self.search_bar.setPlaceholderText("Поиск функций...")
        self.layout.addWidget(self.search_bar)
        
        # Контейнер, где лежат твои функции/кнопки
        self.functions_container = QWidget()
        self.functions_layout = QVBoxLayout(self.functions_container)
        self.layout.addWidget(self.functions_container)

        self.function_items = []
        for i in range(10):
            # Важно: используем элементы, которые сами умеют менять тему!

            self.functions_layout.addWidget(lbl)

        # 2. ПРИВЯЗКА ПОИСКА (то, что Клод, скорее всего, снёс)
        self.search_bar.textChanged.connect(self.filter_functions)
        
        self.setWidget(self.view)

    def filter_functions(self, text):

        for item in self.function_items:

            else:
                item.hide()

    search_text = text.lower().strip()
    
    # Проходим по всем твоим карточкам/кнопкам функций
    for card in self.function_cards: 
        # Проверяем, есть ли поисковое слово в названии или описании карточки
        if search_text in card.title.text().lower() or search_text in card.description.text().lower():
            card.setVisible(True)   # Показываем карточку
            card.setEnabled(True)   # Жестко разрешаем по ней кликать!
        else:
            card.setVisible(False)  # Полностью скрываем (она не будет мешать кликам по другим элементам)
